# Deploy FastAPI source to an existing Azure Linux Web App (no Docker).
# Prerequisite: az login

$ResourceGroup = 'rg-dev-webservice'
$WebAppName = 'app-dev-webservice'
$ProjectDir = Join-Path $PSScriptRoot '.'
$ZipFile = Join-Path $env:TEMP "$WebAppName.zip"

# $env:MONGO_URI = 'mongodb+srv://<user>:<password>@<cluster>/<database>?retryWrites=true&w=majority'
if ([string]::IsNullOrWhiteSpace($env:MONGO_URI)) {
    throw 'Set MONGO_URI in the current PowerShell session before deployment.'
}

# The service uses these environment variables on Azure.
# SCM_DO_BUILD_DURING_DEPLOYMENT makes App Service install dependencies from pyproject.toml.
az webapp config appsettings set `
    --resource-group $ResourceGroup `
    --name $WebAppName `
    --settings `
        "MONGO_URI=$env:MONGO_URI" `
        'MONGO_DB=stub_db' `
        'MONGO_COLLECTION=records' `
        'SCM_DO_BUILD_DURING_DEPLOYMENT=true'

# FastAPI is available through fastapi[standard]; App Service forwards HTTP to port 8000.
az webapp config set `
    --resource-group $ResourceGroup `
    --name $WebAppName `
    --startup-file 'python -m uvicorn service:app --host 0.0.0.0 --port 8000'

# Package only this service directory, then trigger a remote Oryx build and deploy it.
if (Test-Path $ZipFile) {
    Remove-Item -LiteralPath $ZipFile
}
Compress-Archive -Path (Join-Path $ProjectDir '*') -DestinationPath $ZipFile -Force

az webapp deploy `
    --resource-group $ResourceGroup `
    --name $WebAppName `
    --src-path $ZipFile `
    --type zip `
    --async false

# Verify the deployed service.
$HostName = az webapp show --resource-group $ResourceGroup --name $WebAppName --query defaultHostName --output tsv
Invoke-RestMethod "https://$HostName/alpha/v1/info"
